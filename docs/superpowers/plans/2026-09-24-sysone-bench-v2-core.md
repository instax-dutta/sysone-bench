# Sysone-bench v2 Core Contracts and Storage Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add reproducible project tooling, strict typed benchmark contracts, canonical full-manifest hashing, and append-only v2 artifact storage without changing legacy records.

**Architecture:** A new `benchmark` package owns pure validation and serialization functions. `run.py` and `compare.py` will call these functions but will not be rewritten until the runner plan. Dataset and result records use explicit schema-versioned JSON/JSONL and exclusive writes.

**Tech Stack:** Python 3.11 and 3.12, `uv`, standard-library `dataclasses`, `hashlib`, `json`, `pathlib`, `pytest`, Ruff, mypy.

**Spec:** `docs/superpowers/specs/2026-09-24-sysone-bench-v2-design.md`

## Global Constraints

- Preserve every existing file under `results/` byte-for-byte.
- Use full SHA-256 digests for v2 manifests and artifacts.
- Preserve question and case order in hashed representations.
- Reject NaN, infinity, missing IDs, extra IDs, and wrong answer types.
- Use dataset split seed `42` and bootstrap seed `42`.
- Do not commit, push, or publish automatically. Leave changes in the worktree for review.
- Never read, print, log, or commit `TYPESAFE_API_KEY`.
- New durable boundaries require their own `AGENTS.md`.
- Final runtime is Python 3.12 on pelican; CI covers Python 3.11 and 3.12.

```python
import hashlib
import json
import math
from pathlib import Path
from typing import Any


def _reject_nonfinite(value: Any) -> None:
    if isinstance(value, float) and not math.isfinite(value):
        raise ValueError("non-finite JSON number")
    if isinstance(value, dict):
        for child in value.values():
            _reject_nonfinite(child)
    elif isinstance(value, (list, tuple)):
        for child in value:
            _reject_nonfinite(child)


def canonical_json(value: object) -> bytes:
    _reject_nonfinite(value)
    return json.dumps(
        value,
        ensure_ascii=False,
        allow_nan=False,
        sort_keys=True,
        separators=(",", ":"),
    ).encode("utf-8")


def digest_bytes(value: bytes) -> str:
    return hashlib.sha256(value).hexdigest()


def digest_value(value: object) -> str:
    return digest_bytes(canonical_json(value))


def digest_file(path: Path) -> str:
    return digest_bytes(path.read_bytes())
```

Implement exact checks for:

```python
from collections.abc import Mapping, Sequence
from typing import Any


def _probability(value: float) -> float:
    if not isinstance(value, (int, float)) or not 0 <= value <= 1:
        raise ValueError("probability must be finite and in [0, 1]")
    return float(value)


def validate_answer(question: Mapping[str, Any], answer: Mapping[str, Any]) -> None:
    qid = question["qid"]
    if answer.get("type") != question["type"]:
        raise ValueError(f"{qid}: answer type does not match question type")
    if question["type"] == "choice":
        labels = set(question["criteria"])
        if answer.get("choice") not in labels:
            raise ValueError(f"{qid}: illegal choice label")
        probabilities = answer.get("probabilities")
        if not isinstance(probabilities, Mapping) or set(probabilities) != labels:
            raise ValueError(f"{qid}: probability labels do not match choices")
        values = [_probability(probabilities[label]) for label in labels]
        if abs(sum(values) - 1.0) > 1e-6:
            raise ValueError(f"{qid}: probabilities must sum to one")
        if not 0 <= float(answer.get("confidence", -1)) <= 1:
            raise ValueError(f"{qid}: confidence must be in [0, 1]")
    elif question["type"] == "noul":
        _probability(answer.get("noul"))
    elif question["type"] == "score":
        value = float(answer["score"])
        if not 0 <= value <= float(question["max_score"]):
            raise ValueError(f"{qid}: score is outside the declared range")
        if answer.get("confidence") is not None:
            if not 0 <= float(answer["confidence"]) <= 1:
                raise ValueError(f"{qid}: confidence must be in [0, 1]")
    else:
        raise ValueError(f"{qid}: unsupported answer type")


def validate_answers(questions: Sequence[Mapping[str, Any]], answers: Mapping[str, Any]) -> None:
    expected = {question["qid"] for question in questions}
    actual = set(answers)
    if actual != expected:
        raise ValueError(f"answer IDs differ: missing={sorted(expected - actual)}, extra={sorted(actual - expected)}")
    for question in questions:
        validate_answer(question, answers[question["qid"]])
```
